import socket
import math
import time

IP = "10.26.10.20"
PORTA = 30002

RAIO = 0.1
PONTOS = 500

V = 1      # 800 mm/s
A = 1.500       # 1,5 m/s²
BLEND = 0.008   # 6 mm

linhas = []

linhas.append("def circulo_rapido():")
linhas.append("  p0 = get_actual_tcp_pose()")

# Vai até a borda primeiro
linhas.append(
    f"  movel("
    f"p[p0[0]+{RAIO},p0[1],p0[2],p0[3],p0[4],p0[5]],"
    f"a={A},v={V})"
)

# Circunferência
for i in range(1, PONTOS + 1):

    ang = 2.0 * math.pi * i / PONTOS

    dx = RAIO * math.cos(ang)
    dy = RAIO * math.sin(ang)

    pose = (
        f"p["
        f"p0[0]+({dx:.9f}),"
        f"p0[1]+({dy:.9f}),"
        f"p0[2],"
        f"p0[3],"
        f"p0[4],"
        f"p0[5]"
        f"]"
    )

    if i < PONTOS:

        linhas.append(
            f"  movel("
            f"{pose},"
            f"a={A},"
            f"v={V},"
            f"r={BLEND})"
        )

    else:

        linhas.append(
            f"  movel("
            f"{pose},"
            f"a={A},"
            f"v={V})"
        )

# Retorna ao centro
linhas.append(
    f"  movel(p0,a={A},v={V})"
)

linhas.append("end")

script = "\n".join(linhas) + "\n"

print("=" * 50)
print("UR5 - CIRCULO RAPIDO")
print("=" * 50)

print(f"Diâmetro:    {RAIO*2000:.0f} mm")
print(f"Pontos:      {PONTOS}")
print(f"Velocidade:  {V*1000:.0f} mm/s")
print(f"Aceleração:  {A*1000:.0f} mm/s²")
print(f"Blend:       {BLEND*1000:.0f} mm")

input("\nENTER para executar...")

sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
sock.settimeout(5)

try:

    sock.connect((IP, PORTA))
    sock.sendall(script.encode("utf-8"))

    print("Movimento enviado.")

    time.sleep(15)

finally:

    sock.close()
